class Arrow3D(FancyArrowPatch):
    def __init__(self, xs, ys, zs, *args, **kwargs):
        FancyArrowPatch.__init__(self, (0, 0), (0, 0), *args, **kwargs)
        self._verts3d = xs, ys, zs

    def draw(self, renderer):
        xs3d, ys3d, zs3d = self._verts3d
        xs, ys, zs = proj3d.proj_transform(xs3d, ys3d, zs3d, renderer.M)
        self.set_positions((xs[0], ys[0]), (xs[1], ys[1]))
        FancyArrowPatch.draw(self, renderer)

    def do_3d_projection(self, renderer=None):
        xs3d, ys3d, zs3d = self._verts3d
        xs, ys, zs = proj3d.proj_transform(xs3d, ys3d, zs3d, self.axes.M)
        self.set_positions((xs[0], ys[0]), (xs[1], ys[1]))
        return np.min(zs)

def analyze_matrix(A):
    """Compute eigenvalues, Gershgorin bounds, spectral radius."""
    n = A.shape[0]
    
    # Eigenvalues and eigenvectors
    eigenvalues, eigenvectors = eig(A)
    
    # Gershgorin discs
    diag = np.diag(A)
    row_sums = np.sum(np.abs(A), axis=1)
    radii = row_sums - np.abs(diag)
    
    # Spectral radius
    spectral_radius = np.max(np.abs(eigenvalues))
    
    # Gershgorin bound
    gershgorin_bound = np.max(diag + radii)
    
    return {
        'matrix': A,
        'eigenvalues': eigenvalues,
        'eigenvectors': eigenvectors.T,  # Transpose for row-wise access
        'diagonal': diag,
        'radii': radii,
        'spectral_radius': spectral_radius,
        'gershgorin_bound': gershgorin_bound
    }

def neon_color(value, vmin, vmax, cmap_name=NEON_COLORMAP):
    """Map value to neon color."""
    normalized = np.clip((value - vmin) / (vmax - vmin + 1e-10), 0, 1)
    cmap = plt.cm.get_cmap(cmap_name)
    return cmap(normalized)

def neon_eigen(v):
    return neon_color(v, min_rho, max_rho)

def neon_shell(v):
    return neon_color(v, min_gersh, max_gersh)

def neon_fiber(v):
    return neon_color(v, 0, 1)

def neon_tensor(v):
    return neon_color(v, 0, 1)

def neon_tube(v):
    return neon_color(v, min_rho, max_rho)

def compute_composite(result, epoch):
    """Compute all geometric elements for one epoch."""
    A = result['matrix']
    eigenvalues = result['eigenvalues']
    eigenvectors = result['eigenvectors']
    radii = result['radii']
    rho = result['spectral_radius']
    gersh = result['gershgorin_bound']
    
    # Limit eigenvalues if specified
    if MAX_EIGENVALUES is not None:
        n = min(len(eigenvalues), MAX_EIGENVALUES)
        eigenvalues = eigenvalues[:n]
        eigenvectors = eigenvectors[:n]
        radii = radii[:n]
    else:
        n = len(eigenvalues)
    
    # Eigenvalue constellation points
    re_vals = np.real(eigenvalues)
    im_vals = np.imag(eigenvalues)
    
    pts = np.column_stack([re_vals, im_vals, radii])
    
    # Center the constellation
    center_shift = np.mean(pts, axis=0)
    pts = pts - center_shift
    
    # Spheres (position, radius)
    spheres = [(pts[i], radii[i]) for i in range(n)]
    
    # Fiber halo (FULL COMPLEXITY)
    fibers = []
    for i in range(n):
        p = pts[i]
        vec = eigenvectors[i]
        
        # Take first 3 components
        vec3 = vec[:min(3, len(vec))]
        vec3 = np.pad(vec3, (0, max(0, 3 - len(vec3))), constant_values=0)
        vec3 = np.real(vec3)
        
        # Direction
        if np.linalg.norm(vec3) == 0:
            direction = np.array([0, 0, 1])
        else:
            direction = vec3 / np.linalg.norm(vec3)
        
        # Create fiber halo (12 fibers)
        for t in np.linspace(0, 2*np.pi, FIBERS_PER_POINT, endpoint=False):
            perturb = direction + 0.2 * np.array([np.cos(t), np.sin(t), 0])
            perturb = perturb / np.linalg.norm(perturb)
            end_point = p + 0.15 * perturb
            fibers.append((p, end_point))
    
    # Tensor arrows (from A matrix rows)
    tensor_arrows = []
    tensor_rows = A[:TENSOR_ARROW_COUNT, :min(3, A.shape[1])]
    
    for i, row in enumerate(tensor_rows):
        base = np.array([i, -1.5, 0])
        row_padded = np.pad(row, (0, max(0, 3 - len(row))), constant_values=0)
        
        if np.linalg.norm(row_padded) == 0:
            direction = np.array([0, 0, 1])
        else:
            direction = row_padded / np.linalg.norm(row_padded)
        
        end = base + 0.3 * direction
        tensor_arrows.append((base, end))
    
    return {
        'points': pts,
        'spheres': spheres,
        'fibers': fibers,
        'tensor_arrows': tensor_arrows,
        'rho': rho,
        'gersh': gersh,
        'epoch': epoch
    }

composites = [compute_composite(results[i], i+1) for i in range(len(results))]

def stability_tubes(epoch, rho, gersh):
    """Create stability indicator tubes."""
    e = epoch
    
    # Rho tube
    rho_start = np.array([e, 2.0, 0])
    rho_height = 0.2 * (rho - min_rho) / (max_rho - min_rho + 1e-10)
    rho_end = np.array([e + 0.1, 2.0 + rho_height, 0])
    
    # Gershgorin tube
    gersh_start = np.array([e, 2.0, -0.5])
    gersh_height = 0.2 * (gersh - min_gersh) / (max_gersh - min_gersh + 1e-10)
    gersh_end = np.array([e + 0.1, 2.0 + gersh_height, -0.5])
    
    return {
        'rho': (rho_start, rho_end),
        'gersh': (gersh_start, gersh_end)
    }

    for c in composites
]

THRESHOLD = 0.01
logic_maps = [np.sign(np.clip(r['matrix'], -THRESHOLD, THRESHOLD)) for r in results]
LOGIC_SHELL_RADIUS = 3.0

def logic_shell_points(logic_matrix, mode):
    """Generate points on spherical shell colored by logic values."""
    n = logic_matrix.shape[0]
    
    # High-resolution sampling
    resolution = LOGIC_SHELL_RESOLUTION
    
    coords = []
    vals = []
    
    for i in range(resolution):
        theta = i / resolution * np.pi
        for j in range(resolution):
            phi = j / resolution * 2 * np.pi
            
            x = LOGIC_SHELL_RADIUS * np.sin(theta) * np.cos(phi)
            y = LOGIC_SHELL_RADIUS * np.sin(theta) * np.sin(phi)
            z = LOGIC_SHELL_RADIUS * np.cos(theta)
            
            coords.append([x, y, z])
            
            # Map to matrix position
            row = int((i / resolution) * n)
            col = int((j / resolution) * n)
            row = min(row, n-1)
            col = min(col, n-1)
            
            vals.append(logic_matrix[row, col])
    
    return np.array(coords), np.array(vals)

def create_frame(composite_idx):
    """Create a single 3D visualization frame."""
    c = composites[composite_idx]
    tubes = tubes_per_epoch[composite_idx]
    logic = logic_maps[composite_idx]
    
    pts = c['points']
    spheres = c['spheres']
    fibers = c['fibers']
    tensor_arrows = c['tensor_arrows']
    rho = c['rho']
    
    # Hue shift for animation
    hue_shift = (0.1 * composite_idx) % 1.0
    
    # Logic shell
    coords, vals = logic_shell_points(logic, LOGIC_SHELL_MODE)
    
    # Create figure
    fig = plt.figure(figsize=(IMAGE_SIZE[0]/DPI, IMAGE_SIZE[1]/DPI), dpi=DPI)
    ax = fig.add_subplot(111, projection='3d')
    ax.set_facecolor((0.02, 0.02, 0.05))
    fig.patch.set_facecolor((0.02, 0.02, 0.05))
    
    # Plot logic shell
    for k in range(len(coords)):
        val = vals[k]
        if val == 1:
            color = neon_shell(1.0)
        elif val == -1:
            color = neon_shell(0.0)
        else:
            continue  # Skip zero values
        
        if LOGIC_SHELL_MODE == 1:
            # Single layer
            ax.scatter(*coords[k], c=[color], s=5, alpha=0.12)
        elif LOGIC_SHELL_MODE == 2:
            # Double layer
            ax.scatter(*coords[k], c=[color], s=5, alpha=0.12)
            ax.scatter(*(1.05 * coords[k]), c=[color], s=5, alpha=0.12)
        elif LOGIC_SHELL_MODE == 3:
            # Volumetric
            ax.scatter(*coords[k], c=[color], s=3, alpha=0.08)
            ax.scatter(*(1.03 * coords[k]), c=[color], s=3, alpha=0.08)
            ax.scatter(*(1.06 * coords[k]), c=[color], s=3, alpha=0.08)
    
    # Plot Gershgorin spheres
    for sphere_pos, radius in spheres:
        u = np.linspace(0, 2 * np.pi, 20)
        v = np.linspace(0, np.pi, 10)
        x = sphere_pos[0] + radius * np.outer(np.cos(u), np.sin(v))
        y = sphere_pos[1] + radius * np.outer(np.sin(u), np.sin(v))
        z = sphere_pos[2] + radius * np.outer(np.ones(np.size(u)), np.cos(v))
        
        color = neon_shell(radius)
        ax.plot_surface(x, y, z, color=color, alpha=0.10, shade=False)
    
    # Plot eigenvalue points
    color = neon_eigen(rho)
    ax.scatter(pts[:, 0], pts[:, 1], pts[:, 2], 
               c=[color]*len(pts), s=50, alpha=0.30)
    
    # Plot fiber halo
    for start, end in fibers:
        arrow = Arrow3D([start[0], end[0]], 
                       [start[1], end[1]], 
                       [start[2], end[2]],
                       mutation_scale=10, lw=1.5, 
                       arrowstyle='->', 
                       color=neon_fiber(0.7), 
                       alpha=0.45)
        ax.add_artist(arrow)
    
    # Plot tensor arrows
    for start, end in tensor_arrows:
        arrow = Arrow3D([start[0], end[0]], 
                       [start[1], end[1]], 
                       [start[2], end[2]],
                       mutation_scale=15, lw=2, 
                       arrowstyle='->', 
                       color=neon_tensor(0.3), 
                       alpha=0.20)
        ax.add_artist(arrow)
    
    # Plot stability tubes
    rho_tube = tubes['rho']
    gersh_tube = tubes['gersh']
    
    ax.plot([rho_tube[0][0], rho_tube[1][0]],
            [rho_tube[0][1], rho_tube[1][1]],
            [rho_tube[0][2], rho_tube[1][2]],
            color=neon_tube(rho), lw=8, alpha=0.20)
    
    ax.plot([gersh_tube[0][0], gersh_tube[1][0]],
            [gersh_tube[0][1], gersh_tube[1][1]],
            [gersh_tube[0][2], gersh_tube[1][2]],
            color=neon_tube(rho), lw=8, alpha=0.20)
    
    # Set view
    ax.set_xlim([-4, 4])
    ax.set_ylim([-4, 4])
    ax.set_zlim([-4, 4])
    ax.view_init(elev=20, azim=45 + composite_idx * 10)
    
    # Remove axes
    ax.set_axis_off()
    
    plt.tight_layout()
    
    return fig
